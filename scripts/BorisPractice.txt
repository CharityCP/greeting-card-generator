A COUNTDOWN that stops at zero
set an if statement

let count = 10;
let countText = document.getElementById("count-1");

let ticket = setInterval(function () {
  count = count - 1;
  countText.textContent = count;
  // Add an if: when count hits 0, clearInterval(ticket) and say something
  if (count === 0) {
    clearInterval (ticket);
    alert("Time's Up!");
  }
}, 1000);
